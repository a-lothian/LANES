(module
  (memory (export "memory") 1)
  (func $scale (param $s f32) (param $k f32) (result f32)
    local.get $s
    local.get $k
    f32.mul
    return
  )
  (func $pass (param $x i32) (result i32)
    (local $y i32)
    i32.const 0
    local.get $x
    i32.const 8
    memory.copy
    i32.const 0
    local.set $y
    local.get $y
    return
  )
  (func $main (result i32)
    (local $a i32)
    (local $b f32)
    i32.const 16
    i32.const 32
    f32.const 1
    f32.store
    i32.const 32
    f32.const 2
    f32.store offset=4
    i32.const 32
    call $pass
    i32.const 8
    memory.copy
    i32.const 16
    local.set $a
    f32.const 2
    f32.const 3
    call $scale
    local.set $b
    i32.const 0
    return
  )
  (export "scale" (func $scale))
  (export "pass" (func $pass))
  (export "main" (func $main))
)
