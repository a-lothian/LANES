(module
  (func $sum_to (param $n i32) (result i32)
    (local $i i32)
    (local $acc i32)
    i32.const 0
    local.set $i
    i32.const 0
    local.set $acc
    block $while_exit_0
      loop $while_loop_0
        local.get $i
        local.get $n
        i32.lt_s
        i32.eqz
        br_if $while_exit_0
        local.get $acc
        local.get $i
        i32.add
        local.set $acc
        local.get $i
        i32.const 1
        i32.add
        local.set $i
        br $while_loop_0
      end
    end
    local.get $acc
    return
  )
  (export "sum_to" (func $sum_to))
)
