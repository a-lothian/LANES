(module
  (func $sum (param $n i32) (result i32)
    (local $i i32)
    (local $acc i32)
    block $done
      loop $again
        local.get $i
        local.get $n
        i32.ge_s
        br_if $done
        local.get $acc
        local.get $i
        i32.add
        local.set $acc
        local.get $i
        i32.const 1
        i32.add
        local.set $i
        br $again
      end
    end
    local.get $acc
  )
)
