function fact(n: integer) -> integer {
    if n <= 1 {
        return 1;
    }
    return n * fact(n - 1);
}

function main() -> integer {
    let i = 1;
    while i <= 5 {
        print(fact(i));
        i = i + 1;
    }

    if i == 6 {
        print(true);
    } else {
        print(false);
    }
    return 0;
}
